cmake_minimum_required(VERSION 3.16)
project(rk3506_driver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(rk3506_driver_node src/driver.cpp)
target_compile_features(rk3506_driver_node PRIVATE cxx_std_17)
target_compile_options(rk3506_driver_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(rk3506_driver_node rclcpp rcl_interfaces geometry_msgs nav_msgs std_srvs std_msgs sensor_msgs diagnostic_msgs tf2_ros)
install(TARGETS rk3506_driver_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(driver_integration test/test_driver.py TIMEOUT 180
    ENV RK3506_DRIVER_EXECUTABLE=$<TARGET_FILE:rk3506_driver_node>
    ROS_DOMAIN_ID=173 ROS_LOCALHOST_ONLY=1 PYTEST_ADDOPTS=--tb=short)
endif()
ament_package()
